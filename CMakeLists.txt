cmake_minimum_required(VERSION 3.19)
project(s1pro LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
add_executable(s1pro src/main.cpp src/radio.cpp src/radio.h src/tables.cpp src/tables.h)
target_compile_options(s1pro PRIVATE -Wall -Wextra)
target_link_libraries(s1pro PRIVATE Qt6::Widgets Qt6::DBus)
